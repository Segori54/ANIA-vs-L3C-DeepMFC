cmake_minimum_required(VERSION 3.22)
project(PatelReferenceComparator VERSION 1.0 LANGUAGES C CXX)
get_filename_component(WORKSPACE "${CMAKE_CURRENT_SOURCE_DIR}/../.." ABSOLUTE)
set(ESTIMATOR_CPP "${WORKSPACE}/RealtimeFeedbackEngine/Source/Processors/PatelEstimator.cpp")
set(ESTIMATOR_HPP "${WORKSPACE}/RealtimeFeedbackEngine/Source/Processors/PatelEstimator.h")
set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS "${ESTIMATOR_CPP}" "${ESTIMATOR_HPP}")
file(SHA256 "${ESTIMATOR_CPP}" ESTIMATOR_CPP_SHA256)
file(SHA256 "${ESTIMATOR_HPP}" ESTIMATOR_HPP_SHA256)
configure_file(build_info.h.in "${CMAKE_BINARY_DIR}/generated/build_info.h" @ONLY)
set(JUCE_SOURCE_PATH "${WORKSPACE}/build/_deps/juce-src" CACHE PATH "JUCE 8.0.10 sources")
set(JUCE_MODULES_ONLY ON CACHE BOOL "Build only JUCE modules" FORCE)
if(EXISTS "${JUCE_SOURCE_PATH}/CMakeLists.txt")
    file(STRINGS "${JUCE_SOURCE_PATH}/CMakeLists.txt" JUCE_VERSION_LINE REGEX "project\\(JUCE VERSION 8\\.0\\.10 ")
    if(NOT JUCE_VERSION_LINE)
        message(FATAL_ERROR "JUCE_SOURCE_PATH must contain JUCE 8.0.10")
    endif()
    add_subdirectory("${JUCE_SOURCE_PATH}" "${CMAKE_BINARY_DIR}/juce")
else()
    include(FetchContent)
    FetchContent_Declare(JUCE GIT_REPOSITORY https://github.com/juce-framework/JUCE.git GIT_TAG 8.0.10 GIT_SHALLOW TRUE)
    FetchContent_MakeAvailable(JUCE)
endif()
add_executable(patel_compare main.cpp "${WORKSPACE}/RealtimeFeedbackEngine/Source/Processors/PatelEstimator.cpp")
target_include_directories(patel_compare PRIVATE "${WORKSPACE}/RealtimeFeedbackEngine/Source/Processors")
target_include_directories(patel_compare PRIVATE "${CMAKE_BINARY_DIR}/generated")
target_compile_features(patel_compare PRIVATE cxx_std_20)
if(MSVC)
    target_compile_options(patel_compare PRIVATE /EHsc)
endif()
target_compile_definitions(patel_compare PRIVATE JUCE_USE_CURL=0 JUCE_WEB_BROWSER=0)
target_link_libraries(patel_compare PRIVATE juce::juce_dsp)
set_target_properties(patel_compare PROPERTIES RUNTIME_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/bin")
